# Viola Libre
# Cifras, acordes e teoria musical livre para viola caipira, violão e cavaquinho.
#
# O acervo de cifras (/cifras) é público e DEVE ser rastreável: é o que faz alguém
# chegar aqui pela busca. Já esteve bloqueado por engano, e o efeito era o site
# inteiro fora do índice do Google.
#
# Atenção: a Cloudflare injeta um bloco `User-agent: *` próprio antes deste. Quando
# dois grupos casam com o mesmo robô, eles são FUNDIDOS e vence a regra de caminho
# mais longo — então um `Disallow:` aqui derruba o `Allow: /` de lá. Bloqueie apenas
# o que realmente não deve ser indexado.

User-agent: *

# Rascunho local do usuário: vive no navegador dele, não existe no servidor e não
# tem conteúdo público algum a oferecer.
Disallow: /minhascifras

# Ferramentas de edição, não conteúdo. São a mesma cifra que já está indexada em
# /cifras/artista/musica, só que numa tela de trabalho — indexar as duas seria
# conteúdo duplicado.
#
# O `*/*/` casa os DOIS segmentos de /cifras/artista/musica antes do sufixo. Com um
# `*` só, `/cifras/*/timing` pegava também /cifras/artista/timing — e o acervo tem
# seis cifras chamadas «Timing» e «Print», que são páginas legítimas e estavam sendo
# bloqueadas por engano. Quem achou foi o `npm run sitemap:verify`, quando o sitemap
# cresceu o bastante para incluí-las.
Disallow: /cifras/*/*/timing

# Folha de impressão: mesma cifra de /cifras/artista/musica, só que formatada para papel.
Disallow: /cifras/*/*/print

# Grafo harmônico: uma LEITURA da cifra que já está indexada, não conteúdo próprio.
# Uma página fina por cifra, multiplicada pelo acervo, gasta orçamento de rastreio à
# toa — e o relatório de cobertura já mostrava 166 páginas em "rastreada, mas não
# indexada". Bloquear aqui é seguro porque nenhuma delas chegou a ser indexada: elas
# também carregam `noindex` (GrafoPage.tsx), que é o que vale para as que o Google
# porventura já tenha visto.
Disallow: /cifras/*/*/grafo

# Resultado de busca interna: espaço de rastreamento infinito (uma URL por termo
# digitado) e nenhuma delas tem conteúdo próprio — só recorta o que já está
# indexado. `modo` é estado de UI, mesma página.
Disallow: /*?*busca=
Disallow: /*?*modo=

Sitemap: https://violalibre.com.br/sitemap.xml
